### 使用方式

```python
from ai_test_asset_center.discovery_engine import AutonomousDiscoveryEngine
from ai_test_asset_center.analyzers import (
    BusinessRulesAnalyzer,
    StateMachineAnalyzer,
    MultiTenantAnalyzer,
    ConservationAnalyzer,
    # ... 更多分析器
)

# 创建引擎
engine = AutonomousDiscoveryEngine()

# 注册增强分析器
engine.register_analyzer(BusinessRulesAnalyzer())
engine.register_analyzer(StateMachineAnalyzer())
engine.register_analyzer(MultiTenantAnalyzer())
engine.register_analyzer(ConservationAnalyzer())

# 运行发现
findings = engine.run_discovery(prd_text, api_spec_text)

# 增强的发现结果包含更多bug分类
for finding in findings:
    print(f"分类: {finding.categories}")
    print(f"严重性: {finding.severity}")
    print(f"证据: {finding.evidence}")
```
